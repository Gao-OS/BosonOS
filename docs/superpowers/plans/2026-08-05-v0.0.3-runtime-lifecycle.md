# BosonOS v0.0.3 Runtime Lifecycle Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make BEAM readiness trustworthy and let the runtime request one-shot reboot or poweroff actions from minimal Gluon PID 1, with dual-architecture QEMU proof and a published v0.0.3 release.

**Architecture:** `Boson.Boot` becomes a named lifecycle coordinator that monitors the required OTP components and exposes stable status. `Boson.SystemAction` atomically writes `/run/boson/action`; after a clean runtime exit Gluon consumes that bounded file and selects the requested machine operation, while crash status always wins. Nix builds a private poweroff image for lifecycle smoke tests without changing normal release images.

**Tech Stack:** Elixir 1.18 / OTP 27, Zig 0.15, Nix flakes, Linux/QEMU, GitHub Actions.

```elixir
defmodule Boson.SystemAction do
  @moduledoc false

  @default_path "/run/boson/action"
  @max_bytes 16
  @payloads %{reboot: "reboot\n", poweroff: "poweroff\n"}

  def default_path, do: @default_path

  def write(action, path \\ @default_path) do
    case Map.fetch(@payloads, action) do
      :error -> {:error, :unsupported_action}
      {:ok, payload} -> write_payload(path, payload)
    end
  end

  def read(path \\ @default_path) do
    case File.open(path, [:read, :binary]) do
      {:error, :enoent} ->
        :missing

      {:error, reason} ->
        {:error, reason}

      {:ok, io} ->
        result =
          case IO.binread(io, @max_bytes + 1) do
            :eof -> {:error, :invalid_action}
            {:error, reason} -> {:error, reason}
            payload when byte_size(payload) > @max_bytes -> {:error, :too_large}
            "reboot\n" -> {:ok, :reboot}
            "poweroff\n" -> {:ok, :poweroff}
            _payload -> {:error, :invalid_action}
          end

        File.close(io)
        result
    end
  end

  defp write_payload(path, payload) do
    temp = "#{path}.#{System.unique_integer([:positive, :monotonic])}.tmp"

    with :ok <- File.mkdir_p(Path.dirname(path)),
         :ok <- write_temp(temp, payload),
         :ok <- File.chmod(temp, 0o600),
         :ok <- File.rename(temp, path) do
      :ok
    else
      {:error, _reason} = error ->
        File.rm(temp)
        error
    end
  end

  defp write_temp(path, payload) do
    case File.open(path, [:write, :binary, :exclusive]) do
      {:error, reason} ->
        {:error, reason}

      {:ok, io} ->
        result =
          with :ok <- IO.binwrite(io, payload),
               :ok <- :file.sync(io) do
            :ok
          end

        File.close(io)
        result
    end
  end
end
```

```zig
pub fn clearStale(action_path: []const u8) !void {
    std.fs.deleteFileAbsolute(action_path) catch |err| switch (err) {
        error.FileNotFound => return,
        else => return err,
    };
}

pub fn consume(action_path: []const u8) FileState {
    var file = std.fs.openFileAbsolute(action_path, .{}) catch |err| switch (err) {
        error.FileNotFound => return .missing,
        else => {
            clearStale(action_path) catch return .remove_error;
            return .read_error;
        },
    };
    defer file.close();

    var buffer: [max_bytes + 1]u8 = undefined;
    const count = file.readAll(&buffer) catch {
        clearStale(action_path) catch return .remove_error;
        return .read_error;
    };
    const requested = if (count > max_bytes) null else parse(buffer[0..count]);

    clearStale(action_path) catch return .remove_error;
    return if (requested) |value| .{ .requested = value } else .invalid;
}
```

The open descriptor may remain valid while the Linux directory entry is
removed; `defer file.close()` still closes it before returning.
